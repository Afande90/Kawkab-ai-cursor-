cmake_minimum_required(VERSION 3.22)
project(kawkab_decision LANGUAGES CXX)
# Prepare pinned headers and the *existing* Llamatik libraries using prepare_native.py.
set(INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/prepared")
foreach(LIB llama ggml ggml-cpu ggml-base omp)
    add_library(${LIB} SHARED IMPORTED)
    set_target_properties(${LIB} PROPERTIES IMPORTED_LOCATION "${INPUTS}/${ANDROID_ABI}/lib${LIB}.so")
endforeach()
add_library(kawkab_verdict STATIC verdict.cpp)
set_target_properties(kawkab_verdict PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(kawkab_verdict PUBLIC "${INPUTS}/include")
target_compile_features(kawkab_verdict PUBLIC cxx_std_17)
target_compile_options(kawkab_verdict PRIVATE -Wall -Wextra -Werror)
target_link_libraries(kawkab_verdict PUBLIC llama ggml ggml-cpu ggml-base omp)
add_library(kawkab_decision SHARED jev.cpp)
target_include_directories(kawkab_decision PRIVATE "${INPUTS}/include")
target_compile_features(kawkab_decision PRIVATE cxx_std_17)
target_compile_options(kawkab_decision PRIVATE -Wall -Wextra -Werror)
target_link_options(kawkab_decision PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(kawkab_decision PRIVATE kawkab_verdict)
add_executable(kawkab_jev_check phone_check.cpp)
target_compile_options(kawkab_jev_check PRIVATE -Wall -Wextra -Werror)
target_link_options(kawkab_jev_check PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(kawkab_jev_check PRIVATE kawkab_verdict)
